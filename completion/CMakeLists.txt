cmake_minimum_required(VERSION 3.16)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(kronuz_completion LANGUAGES CXX)
endif()

find_package(Threads REQUIRED)
add_library(kronuz_completion INTERFACE)
target_include_directories(kronuz_completion INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/..")
target_compile_features(kronuz_completion INTERFACE cxx_std_20)
target_link_libraries(kronuz_completion INTERFACE Threads::Threads)
add_library(kronuz::completion ALIAS kronuz_completion)

# Optional header adapter; the host supplies standalone Asio. No fetch here.
add_library(kronuz_completion_asio INTERFACE)
target_link_libraries(kronuz_completion_asio INTERFACE kronuz::completion)
target_compile_definitions(kronuz_completion_asio INTERFACE ASIO_STANDALONE)
if(ASIO_INCLUDE_DIR)
  target_include_directories(kronuz_completion_asio INTERFACE "${ASIO_INCLUDE_DIR}")
endif()
add_library(kronuz::completion_asio ALIAS kronuz_completion_asio)
